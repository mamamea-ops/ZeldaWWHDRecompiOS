cmake_minimum_required(VERSION 3.20)
project(wwhd_recomp LANGUAGES C CXX OBJCXX)

find_package(Python3 REQUIRED COMPONENTS Interpreter)

# Renderers compiled into the executable:
#   BOTH   (Metal + Vulkan; one app, choose at runtime)
#   METAL  Metal only
#   VULKAN Vulkan only, portable SDL host
set(WWHD_RENDERER "" CACHE STRING "Graphics backends: BOTH, METAL or VULKAN")
if(NOT WWHD_RENDERER)
  if(APPLE)
    set(WWHD_RENDERER BOTH)
  else()
    set(WWHD_RENDERER VULKAN)
  endif()
endif()
string(TOUPPER "${WWHD_RENDERER}" WWHD_RENDERER)
set_property(CACHE WWHD_RENDERER PROPERTY STRINGS BOTH METAL VULKAN)

if(NOT WWHD_RENDERER MATCHES "^(BOTH|METAL|VULKAN)$")
  message(FATAL_ERROR "WWHD_RENDERER must be BOTH, METAL or VULKAN")
endif()

set(WWHD_HAS_METAL OFF)
set(WWHD_HAS_VULKAN OFF)

if(WWHD_RENDERER MATCHES "^(BOTH|METAL)$")
  if(NOT APPLE)
    message(FATAL_ERROR "Metal is available only on macOS; select VULKAN")
  endif()
  set(WWHD_HAS_METAL ON)
  enable_language(OBJCXX)
endif()

if(WWHD_RENDERER MATCHES "^(BOTH|VULKAN)$")
  set(WWHD_HAS_VULKAN ON)
endif()

if(WWHD_HAS_VULKAN AND NOT WWHD_HAS_METAL)
  set(WWHD_SDL_HOST ON)
else()
  set(WWHD_SDL_HOST OFF)
endif()

message(STATUS "WWHD renderers: ${WWHD_RENDERER} (Metal ${WWHD_HAS_METAL}, Vulkan ${WWHD_HAS_VULKAN}, SDL host ${WWHD_SDL_HOST})")

if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang")
  message(FATAL_ERROR "The recompiled game requires Clang (musttail support); select clang/clang++")
endif()

set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()

# Generated by tools/recomp/recomp.py from game/code/cking.rpx
set(GEN_DIR "${CMAKE_SOURCE_DIR}/build/gen" CACHE PATH "recompiler output")
set(RECOMP_SCRIPT "${CMAKE_SOURCE_DIR}/tools/recomp/recomp.py")
set(RECOMP_INPUT "${CMAKE_SOURCE_DIR}/game/code/cking.rpx")

# If the game dump exists, generate the code before build.
if(EXISTS "${RECOMP_INPUT}")
  file(MAKE_DIRECTORY "${GEN_DIR}")
  execute_process(
    COMMAND "${Python3_EXECUTABLE}" "${RECOMP_SCRIPT}" "${RECOMP_INPUT}" "${GEN_DIR}"
    WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
    RESULT_VARIABLE RECOMP_RESULT
    OUTPUT_VARIABLE RECOMP_STDOUT
    ERROR_VARIABLE RECOMP_STDERR
  )
  if(NOT RECOMP_RESULT EQUAL 0)
    message(FATAL_ERROR "Recompilation failed:\n${RECOMP_STDOUT}\n${RECOMP_STDERR}")
  endif()
endif()

file(GLOB GEN_SOURCES CONFIGURE_DEPENDS "${GEN_DIR}/code_*.c")

# Gather engine source files
file(GLOB_RECURSE WWHD_ENGINE_SOURCES CONFIGURE_DEPENDS
  "${CMAKE_SOURCE_DIR}/runtime/src/*.c"
  "${CMAKE_SOURCE_DIR}/runtime/src/*.cc"
  "${CMAKE_SOURCE_DIR}/runtime/src/*.cpp"
  "${CMAKE_SOURCE_DIR}/runtime/src/*.mm"
  "${CMAKE_SOURCE_DIR}/runtime/src/*.m"
)

# iOS build cannot compile macOS-only UI/input files and desktop Metal backend pieces
# that depend on AppKit or Wii U Latte internals unavailable in the iOS SDK.
# Vulkan is also not available on iOS.
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  list(FILTER WWHD_ENGINE_SOURCES EXCLUDE REGEX "/runtime/src/gfx/(menu|input)\\.mm$")
  list(FILTER WWHD_ENGINE_SOURCES EXCLUDE REGEX "/runtime/src/gfx/controls_ui\\.mm$")
  list(FILTER WWHD_ENGINE_SOURCES EXCLUDE REGEX "/runtime/src/gfx/(metal_main|metal_draw|metal_surfaces|overlay_appkit|quit_prompt|shader_headstart|text_input_appkit)\\.mm$")
  list(FILTER WWHD_ENGINE_SOURCES EXCLUDE REGEX "/runtime/src/gfx/vulkan/.*\\.(cpp|h|mm)$")
  list(FILTER WWHD_ENGINE_SOURCES EXCLUDE REGEX "/runtime/src/gx2/decompiler_glue\\.cpp$")
endif()

# Fallback stub if no engine sources are present (for CI/build scaffolding)
if(NOT WWHD_ENGINE_SOURCES AND NOT GEN_SOURCES)
  message(STATUS "Warning: No generated code found. The app will require user-provided game files at runtime.")
  set(WWHD_ENGINE_STUB "${CMAKE_BINARY_BINARY_DIR}/wwhd_engine_stub.cpp")
  file(WRITE "${WWHD_ENGINE_STUB}" "extern \"C\" int wwhd_engine_stub() { return 0; }\\n")
  set(WWHD_ENGINE_SOURCES "${WWHD_ENGINE_STUB}")
endif()

# Create the target the workflow is trying to build
add_library(wwhd_engine STATIC)

target_sources(wwhd_engine PRIVATE ${WWHD_ENGINE_SOURCES})

if(GEN_SOURCES)
  target_sources(wwhd_engine PRIVATE ${GEN_SOURCES})
endif()

target_include_directories(wwhd_engine
  PUBLIC
    "${CMAKE_SOURCE_DIR}/runtime/include"
    "${CMAKE_SOURCE_DIR}/runtime/src"
    "${GEN_DIR}"
)

target_compile_features(wwhd_engine
  PRIVATE
    cxx_std_20
    c_std_11
)

# iOS-specific framework linkage
if(APPLE)
  target_link_libraries(wwhd_engine
    PUBLIC
      "-framework Foundation"
  )

  if(WWHD_HAS_METAL)
    target_link_libraries(wwhd_engine
      PUBLIC
        "-framework Metal"
        "-framework MetalKit"
    )
  endif()
endif()

# Set up platform-specific flags if needed
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  set_target_properties(wwhd_engine PROPERTIES
    XCODE_ATTRIBUTE_ENABLE_BITCODE NO
    POSITION_INDEPENDENT_CODE ON
  )
endif()
